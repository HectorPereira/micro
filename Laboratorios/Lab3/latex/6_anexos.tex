void move_axis(volatile uint8_t *port_dir, uint8_t dir_bit,
               volatile uint8_t *port_step, uint8_t step_bit,
               uint8_t direction, uint16_t steps)
{
    if (g_limit_hit) return; // detener si hay emergencia

    if(direction) *port_dir |=  (1<<dir_bit);
    else           *port_dir &= ~(1<<dir_bit);
    _delay_loop_2(1);

    for (uint16_t i = 0; i < steps; i++) {
        if (g_limit_hit) break;
        *port_step |=  (1<<step_bit);
        _delay_loop_2(300);
        *port_step &= ~(1<<step_bit);
        _delay_loop_2(300);
    }
}

void move_x(uint8_t dir, float dist_mm) {
    uint16_t steps = (uint16_t)(dist_mm * STEP_SCALE);
    move_axis(&PORTB, DIR_X, &PORTB, STEP_X, dir, steps);
}

void move_y(uint8_t dir, float dist_mm) {
    uint16_t steps = (uint16_t)(dist_mm * STEP_SCALE);
    move_axis(&PORTC, DIR_Y, &PORTC, STEP_Y, dir, steps);
}

\paragraph{Ejecucion de trayectorias programadas}\label{anexo:P5_execute_path}---\\
\begin{lstlisting}[language=C, caption={Ejecucion de trayectorias almacenadas en PROGMEM.}]
void execute_path(const Move *path, uint16_t size, float path_scale) {
    for (uint16_t i = 0; i < size / sizeof(Move); i++) {
        if (g_limit_hit) break;

        Move m;
        m.cmd   = pgm_read_byte(&path[i].cmd);
        m.dx_mm = pgm_read_float(&path[i].dx_mm);
        m.dy_mm = pgm_read_float(&path[i].dy_mm);

        if (m.cmd == 'U') pen_up();
        else if (m.cmd == 'D') pen_down();

        if (m.dx_mm > 0.0f)      move_x(1,  m.dx_mm * path_scale);
        else if (m.dx_mm < 0.0f) move_x(0, -m.dx_mm * path_scale);

        if (m.dy_mm > 0.0f)      move_y(1,  m.dy_mm * path_scale);
        else if (m.dy_mm < 0.0f) move_y(0, -m.dy_mm * path_scale);
    }
}
\end{lstlisting}

void execute_path_mirrored(const Move *path, uint16_t size, float path_scale) {
    execute_path(path, size, path_scale);

    for (uint16_t i = size / sizeof(Move); i > 0; i--) {
        if (g_limit_hit) break;

        Move m;

\paragraph{Inicializacion de ADC y PWM}\label{anexo:T1_adc_pwm_init}---\\
\begin{lstlisting}[language=C, caption={Inicializacion del ADC y del PWM para el control termico.}]
#define F_CPU 16000000UL
#include <avr/io.h>
#include <util/delay.h>
#include <avr/interrupt.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void Init_ADC(void){
    ADMUX  = 0b01000001;                        // AVcc como referencia, canal ADC1
    ADCSRA = (1 << ADEN) | (1 << ADPS2) |
             (1 << ADPS1) | (1 << ADPS0);       // Prescaler 128
    DIDR0  = (1 << ADC1D);                      // Deshabilitar entrada digital
}

void Init_PWM(void){
    DDRD  |= (1 << DDD6);                       // PD6 como salida (OC0A)
    TCCR0A = (1 << COM0A1) | (1 << WGM00);      // Fast PWM, non-inverting
    TCCR0B = (1 << CS01) | (1 << CS00);         // Prescaler 64
    OCR0A = 0;                                  // Ciclo inicial 0%
}
\end{lstlisting}

\paragraph{Lectura bloqueante del ADC}\label{anexo:T2_adc_read_blocking}---\\
\begin{lstlisting}[language=C, caption={Funcion de lectura bloqueante del ADC utilizando el flag ADIF.}]
uint16_t adc_read_blocking_adif(uint8_t canal){
    ADMUX = (ADMUX & 0xF0) | (canal & 0x0F); // Seleccionar canal
    ADCSRA |= (1 << ADSC);                   // Iniciar conversion
    while(!(ADCSRA & (1 << ADIF)));          // Esperar fin
    ADCSRA |= (1 << ADIF);                   // Limpiar bandera
    return ADC;                              // Retornar resultado
}
\end{lstlisting}

\paragraph{Control PWM del ventilador}\label{anexo:T3_control_pwm_fan}---\\
\begin{lstlisting}[language=C, caption={Control de velocidad del ventilador segun la temperatura.}]
void Control_Ventilador(uint16_t valor_adc, uint16_t tmin, uint16_t tmax){
    float temp = (valor_adc * 5.0 / 1023.0) * 100.0; // Conversion LM35
    if (temp < tmin){
        OCR0A = 0;          // Ventilador apagado
    }
    else if (temp > tmax){
        OCR0A = 255;        // Maxima velocidad
    }
    else{
        OCR0A = (uint8_t)(255.0 * (temp - tmin) / (tmax - tmin));
    }
}
\end{lstlisting}

#include <util/delay.h>
#include <avr/interrupt.h>
#include <string.h>

#define TX_BUF_SZ 256
#define RX_BUF_SZ 256
#define BAUD_RATE 9600
#define IN1 PD2
#define IN2 PD3

void usart_init(void) {
    const uint16_t ubrr = (16000000UL / (16UL * BAUD_RATE)) - 1;
    UBRR0H = ubrr >> 8;
    UBRR0L = ubrr;
    UCSR0B = (1 << TXEN0) | (1 << RXEN0) | (1 << RXCIE0);
    UCSR0C = (1 << UCSZ01) | (1 << UCSZ00);
}

void adc_init(void) {
    ADMUX  = (1 << REFS0);                        
    ADCSRA = (1 << ADEN) | (1 << ADPS2) |
             (1 << ADPS1) | (1 << ADPS0);          
}

void setup_pwm_and_dir(void) {
    DDRD |= (1 << IN1) | (1 << IN2);
    DDRB |= (1 << PORTB1);
    TCCR1A = (1 << COM1A1) | (1 << WGM10) | (1 << WGM11);
    TCCR1B = (1 << WGM12) | (1 << CS12); // clk/8

\paragraph{Funciones auxiliares de comunicacion y conversion}\label{anexo:M2_usart_utils}---\\
\begin{lstlisting}[language=C, caption={Rutinas auxiliares para conversion y manejo de buffers USART.}]
uint8_t tx_buf[TX_BUF_SZ];
uint8_t tx_head = 0, tx_tail = 0;
uint8_t rx_buf[RX_BUF_SZ];
uint8_t rx_head = 0, rx_tail = 0;

void UTOA(uint16_t value, char *buffer) { 
    char temp[6];
    int i = 0, j = 0;

    if (value == 0) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return;
    }
    while (value > 0 && i < sizeof(temp) - 1) {
        temp[i++] = (value % 10) + '0';
        value /= 10;
    }
    while (i > 0) buffer[j++] = temp[--i];
    buffer[j] = '\0';
}

uint8_t usart_write_try(uint8_t b) {
    uint8_t next = (uint8_t)((tx_head + 1) & (TX_BUF_SZ - 1));
    if (next == tx_tail) return 0;
    tx_buf[tx_head] = b;
    tx_head = next;
    UCSR0B |= (1 << UDRIE0);
    return 1;
}

uint16_t usart_write_str(const char *s) {
    uint16_t n = 0;
    while (*s && usart_write_try((uint8_t)*s++)) n++;
    return n;
}

uint16_t adc_read(uint8_t channel) {
    ADMUX = (ADMUX & 0xF0) | (channel & 0x0F);
    ADCSRA |= (1 << ADSC);
    while (ADCSRA & (1 << ADSC));
    return ADC;
}
\end{lstlisting}

\paragraph{Lazo de control y envio de telemetria}\label{anexo:M3_usart_task}---\\
\begin{lstlisting}[language=C, caption={Funcion principal de control y retroalimentacion via UART.}]
#define MIN_PWM 150
#define MAX_PWM 600
#define DEAD_ZONE 20

uint16_t motor_pot = 0;
uint16_t reference_pot = 0;

void usart_task(void) {
    uint8_t direction = 0;
    motor_pot = adc_read(3);
    reference_pot = adc_read(4);

    int16_t error = (int16_t)reference_pot - (int16_t)motor_pot;
    uint16_t pwm_output = 0;

    if (error > DEAD_ZONE) {
        PORTD |=  (1 << IN1);
        PORTD &= ~(1 << IN2);
        direction = 1;
        pwm_output = error * 2;
        if (pwm_output < MIN_PWM) pwm_output = MIN_PWM;
        if (pwm_output > MAX_PWM) pwm_output = MAX_PWM;
        OCR1A = pwm_output;
    } 
    else if (error < -DEAD_ZONE) {
        PORTD |=  (1 << IN2);
        PORTD &= ~(1 << IN1);
        pwm_output = (-error) * 2;
        if (pwm_output < MIN_PWM) pwm_output = MIN_PWM;
        if (pwm_output > MAX_PWM) pwm_output = MAX_PWM;
        OCR1A = pwm_output;
    } 
    else {
        PORTD &= ~((1 << IN1) | (1 << IN2));
        OCR1A = 0;
    }

    char buffer[8];
    UTOA(reference_pot, buffer);
    usart_write_str(buffer); usart_write_str(" ");
    UTOA(motor_pot, buffer);
    usart_write_str(buffer); usart_write_str(" ");
    UTOA(pwm_output, buffer);
    usart_write_str(buffer); usart_write_str(" ");
    if (!pwm_output) usart_write_str("Stop\r\n");
    else if (!direction) usart_write_str("Izq.\r\n");
    else usart_write_str("Der.\r\n");
}
\end{lstlisting}

\paragraph{Implementacion del antirrebote temporal}\label{anexo:MA6_debounce_task}---\\
\begin{lstlisting}[language=C, caption={Antirrebote mediante temporizacion usando Timer0 y variables globales.}]
uint32_t millis_counter = 0;
uint32_t debounce_on_at = 0;
uint8_t debounce_active = 0;

ISR(TIMER0_OVF_vect){ millis_counter++; }

uint32_t millis_now(void) {
    uint32_t m;
    cli();
    m = millis_counter;
    sei();
    return m;
}

void debouce_task(void){
    if (debounce_active && (millis_now() >= debounce_on_at)){
        PCIFR |= (1<<PCIF1);
        PCICR |= (1<<PCIE1);
        debounce_active = 0;
    }
}
