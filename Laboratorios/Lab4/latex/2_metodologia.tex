\subsection{Materiales y Herramientas}

\begin{itemize}
    \item \textbf{Microcontrolador principal:}
    \begin{itemize}
        \item ATmega328P
    \end{itemize}

    \item \textbf{Módulos y periféricos electrónicos:}
    \begin{itemize}
        \item Sensor de temperatura LM35
        \item Ventilador DC 5~V
        \item Motor DC + potenciómetro acoplado
        \item Joystick analógico
        \item Botón pulsador
        \item Matriz de LEDs WS2812 (8×8)
        \item Módulo RFID MFRC522
        \item Pantalla LCD 16×2 con interfaz I2C (PCF8574)
        \item Memoria EEPROM interna
        \item LEDs indicadores (rojo, verde)
        \item Buzzer piezoeléctrico
        \item Fuente de alimentación 5~V–12~V
    \end{itemize}

    \item \textbf{Herramientas de programación y simulación:}
    \begin{itemize}
        \item Microchip Studio (Atmel Studio 7)
        \item Proteus 8 / PicSimLab
        \item Python 3.x
        \item Bibliotecas Python: matplotlib, threading, serial (pySerial), collections.deque
    \end{itemize}
\end{itemize}
